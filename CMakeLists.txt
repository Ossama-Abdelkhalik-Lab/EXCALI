cmake_minimum_required(VERSION 3.20)
project(eLam VERSION 0.1.0 LANGUAGES Fortran)
enable_testing()
option(ELAM_OPENMP "Enable OpenMP over independent original batch cases" ON)
if(NOT CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "The original-kernel comparison currently requires GNU Fortran")
endif()
add_library(elam src/elam_solver.f90)
set_target_properties(elam PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
target_include_directories(elam PUBLIC "${CMAKE_BINARY_DIR}/modules")
target_compile_options(elam PRIVATE -O3 -ffree-line-length-none -fno-fast-math)
# Namespace only generated copies; no numerical edits to imported files.
foreach(pair ser vect)
    file(READ "${CMAKE_SOURCE_DIR}/original/exact_${pair}.F90" source)
    if(pair STREQUAL "ser")
        set(module exact_original_serial)
    else()
        set(module exact_original_vector)
    endif()
    string(REPLACE rect_mod_fast "${module}" source "${source}")
    file(WRITE "${CMAKE_BINARY_DIR}/exact_${pair}.F90" "${source}")
endforeach()
add_library(elam_original "${CMAKE_BINARY_DIR}/exact_ser.F90" "${CMAKE_BINARY_DIR}/exact_vect.F90" src/exact_adapter.f90)
set_target_properties(elam_original PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_BINARY_DIR}/original_modules")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/original_modules")
target_include_directories(elam_original PUBLIC "${CMAKE_BINARY_DIR}/original_modules")
target_compile_options(elam_original PRIVATE -O3 -ffree-line-length-none -fno-fast-math)
set_source_files_properties("${CMAKE_BINARY_DIR}/exact_ser.F90" PROPERTIES COMPILE_OPTIONS "-fno-tree-vectorize;-fno-tree-slp-vectorize")
set_source_files_properties("${CMAKE_BINARY_DIR}/exact_vect.F90" PROPERTIES COMPILE_OPTIONS "-fno-trapping-math")
if(ELAM_OPENMP)
    find_package(OpenMP REQUIRED COMPONENTS Fortran)
    target_link_libraries(elam_original PUBLIC OpenMP::OpenMP_Fortran)
endif()
add_executable(test_elam tests/test_elam_standalone.f90)
target_link_libraries(test_elam PRIVATE elam)
target_compile_options(test_elam PRIVATE -ffree-line-length-none)
add_test(NAME elam_truth COMMAND test_elam)
add_executable(test_exact_original tests/test_exact_original.f90)
target_link_libraries(test_exact_original PRIVATE elam_original)
target_compile_options(test_exact_original PRIVATE -ffree-line-length-none)
add_test(NAME original_truth COMMAND test_exact_original)
set_tests_properties(original_truth PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=4")
add_test(NAME original_hashes COMMAND "${CMAKE_COMMAND}" "-DSOURCE_ROOT=${CMAKE_SOURCE_DIR}" -P "${CMAKE_SOURCE_DIR}/cmake/verify_original.cmake")
add_executable(elam_example examples/transfer.f90)
target_link_libraries(elam_example PRIVATE elam)
