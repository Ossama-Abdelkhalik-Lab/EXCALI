cmake_minimum_required(VERSION 3.20)
project(EXCALI VERSION 0.1.0 LANGUAGES Fortran)
enable_testing()
option(EXCALI_OPENMP "Enable OpenMP over independent original batch cases" ON)
if(NOT CMAKE_Fortran_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "The original-kernel comparison currently requires GNU Fortran")
endif()
# Optional profile-guided build of the solver library (about 4% faster):
# configure with EXCALI_PGO=GENERATE, build, run ./excali_pgo_train, then
# reconfigure the same build directory with EXCALI_PGO=USE and rebuild.
set(EXCALI_PGO OFF CACHE STRING "Profile-guided optimization: OFF, GENERATE or USE")
set_property(CACHE EXCALI_PGO PROPERTY STRINGS OFF GENERATE USE)
set(EXCALI_PGO_DIR "${CMAKE_BINARY_DIR}/pgo" CACHE PATH "Profile data directory")
add_library(excali src/excali_solver.f90)
set_target_properties(excali PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/modules")
target_include_directories(excali PUBLIC "${CMAKE_BINARY_DIR}/modules")
target_compile_options(excali PRIVATE -O3 -ffree-line-length-none -fno-fast-math)
if(EXCALI_PGO STREQUAL "GENERATE")
    target_compile_options(excali PRIVATE "-fprofile-generate=${EXCALI_PGO_DIR}")
    target_link_options(excali PUBLIC "-fprofile-generate=${EXCALI_PGO_DIR}")
elseif(EXCALI_PGO STREQUAL "USE")
    target_compile_options(excali PRIVATE "-fprofile-use=${EXCALI_PGO_DIR}" -fprofile-correction -Wno-missing-profile)
elseif(NOT EXCALI_PGO STREQUAL "OFF")
    message(FATAL_ERROR "EXCALI_PGO must be OFF, GENERATE or USE")
endif()
# Namespaced generated copies; the imported files stay unchanged.
foreach(pair ser vect)
    file(READ "${CMAKE_SOURCE_DIR}/original/excali_v1_${pair}.F90" source)
    if(pair STREQUAL "ser")
        set(module excali_v1_serial)
    else()
        set(module excali_v1_vector)
    endif()
    string(REPLACE rect_mod_fast "${module}" source "${source}")
    file(WRITE "${CMAKE_BINARY_DIR}/excali_v1_${pair}.F90" "${source}")
endforeach()
add_library(excali_v1 "${CMAKE_BINARY_DIR}/excali_v1_ser.F90" "${CMAKE_BINARY_DIR}/excali_v1_vect.F90" src/excali_v1_adapter.f90)
set_target_properties(excali_v1 PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_BINARY_DIR}/original_modules")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/original_modules")
target_include_directories(excali_v1 PUBLIC "${CMAKE_BINARY_DIR}/original_modules")
target_compile_options(excali_v1 PRIVATE -O3 -ffree-line-length-none -fno-fast-math)
set_source_files_properties("${CMAKE_BINARY_DIR}/excali_v1_ser.F90" PROPERTIES COMPILE_OPTIONS "-fno-tree-vectorize;-fno-tree-slp-vectorize")
set_source_files_properties("${CMAKE_BINARY_DIR}/excali_v1_vect.F90" PROPERTIES COMPILE_OPTIONS "-fno-trapping-math")
# EXCALI v0 (original/excali_v0.F90) as a plain library: its MEX gateway is
# excluded by -DEXCALI_V0_STANDALONE.
add_library(excali_v0 original/excali_v0.F90 src/excali_v0_adapter.f90)
set_target_properties(excali_v0 PROPERTIES Fortran_MODULE_DIRECTORY "${CMAKE_BINARY_DIR}/v0_modules")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/v0_modules")
target_include_directories(excali_v0 PUBLIC "${CMAKE_BINARY_DIR}/v0_modules")
target_compile_definitions(excali_v0 PRIVATE EXCALI_V0_STANDALONE)
target_compile_options(excali_v0 PRIVATE -O3 -ffree-line-length-none -fno-fast-math)
if(EXCALI_OPENMP)
    find_package(OpenMP REQUIRED COMPONENTS Fortran)
    target_link_libraries(excali_v1 PUBLIC OpenMP::OpenMP_Fortran)
    target_link_libraries(excali_v0 PUBLIC OpenMP::OpenMP_Fortran)
endif()
add_executable(test_excali tests/test_excali_standalone.f90)
target_link_libraries(test_excali PRIVATE excali)
target_compile_options(test_excali PRIVATE -ffree-line-length-none)
add_test(NAME excali_truth COMMAND test_excali)
add_executable(test_excali_v1 tests/test_excali_v1.f90)
target_link_libraries(test_excali_v1 PRIVATE excali_v1)
target_compile_options(test_excali_v1 PRIVATE -ffree-line-length-none)
add_test(NAME excali_v1_truth COMMAND test_excali_v1)
set_tests_properties(excali_v1_truth PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=4")
add_executable(test_excali_v0 tests/test_excali_v0.f90)
target_link_libraries(test_excali_v0 PRIVATE excali_v0)
target_compile_options(test_excali_v0 PRIVATE -ffree-line-length-none)
add_test(NAME excali_v0_truth COMMAND test_excali_v0)
set_tests_properties(excali_v0_truth PROPERTIES ENVIRONMENT "OMP_NUM_THREADS=4")
add_test(NAME original_hashes COMMAND "${CMAKE_COMMAND}" "-DSOURCE_ROOT=${CMAKE_SOURCE_DIR}" -P "${CMAKE_SOURCE_DIR}/cmake/verify_original.cmake")
add_executable(excali_example examples/transfer.f90)
target_link_libraries(excali_example PRIVATE excali)
add_executable(excali_pgo_train examples/pgo_train.f90)
target_link_libraries(excali_pgo_train PRIVATE excali)
